cmake_minimum_required(VERSION 3.20)
project(fox_vs_fly LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(game sim/core/game.cpp sim/core/batch.cpp)
target_include_directories(game PUBLIC sim/include)
target_compile_options(game PRIVATE -Wall -Wextra -Werror)
add_library(neural_rate brain/core/rate.cpp)
target_include_directories(neural_rate PUBLIC brain/include)
if(EMSCRIPTEN)
 add_executable(neural brain/core/rate_wasm.cpp)
 target_link_libraries(neural PRIVATE neural_rate)
 set_target_properties(neural PROPERTIES SUFFIX ".js")
 target_link_options(neural PRIVATE "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sENVIRONMENT=web,node" "-sFILESYSTEM=0" "-sALLOW_MEMORY_GROWTH=1" "-sMAXIMUM_MEMORY=536870912" "-sEXPORTED_RUNTIME_METHODS=['HEAPU32','HEAPU8']" "-sEXPORTED_FUNCTIONS=['_rate_allocate','_rate_offsets','_rate_targets','_rate_weights','_rate_input','_rate_values','_rate_packed','_rate_initialize','_rate_reset','_rate_step','_rate_pack']")
 add_executable(sim sim/core/wasm_api.cpp)
 target_link_libraries(sim PRIVATE game)
 set_target_properties(sim PROPERTIES SUFFIX ".js")
 target_link_options(sim PRIVATE "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sENVIRONMENT=web,node" "-sFILESYSTEM=0" "-sALLOW_MEMORY_GROWTH=0" "-sEXPORTED_FUNCTIONS=['_sim_reset','_sim_step','_sim_field','_sim_tick','_sim_winner','_sim_hash']" "-sASSERTIONS=1")
else()
 enable_testing()
 add_executable(rate_tests brain/tests/rate_tests.cpp)
 target_link_libraries(rate_tests PRIVATE neural_rate)
 add_test(NAME rate COMMAND rate_tests)
 add_executable(sim_tests sim/tests/game_tests.cpp)
 target_link_libraries(sim_tests PRIVATE game)
 add_test(NAME game_tests COMMAND sim_tests)
 add_executable(jump_reference_tests sim/tests/jump_reference.cpp)
 target_link_libraries(jump_reference_tests PRIVATE game)
 add_test(NAME jump_reference COMMAND jump_reference_tests)
 add_executable(sim_replay sim/tests/replay.cpp)
 target_link_libraries(sim_replay PRIVATE game)
 add_test(NAME replay COMMAND sim_replay ${CMAKE_SOURCE_DIR}/sim/tests/fixtures/duel.inputs)
 add_executable(batch_tests sim/tests/batch_tests.cpp)
 target_link_libraries(batch_tests PRIVATE game)
 add_test(NAME batch COMMAND batch_tests)
 add_executable(batch_bench sim/bench/batch_bench.cpp)
 target_link_libraries(batch_bench PRIVATE game)
 add_executable(sim_bench sim/bench/bench.cpp)
 target_link_libraries(sim_bench PRIVATE game)
endif()
